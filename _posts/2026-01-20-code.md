For displaying code in a list item, you have to be aware of the indentation, as stated in [this FAQ](https://github.com/planetjekyll/quickrefs/blob/master/FAQ.md#q-how-can-i-get-backtick-fenced-code-blocks-eg--working-inside-lists-with-kramdown). You must indent your code by **(3 \* bullet_indent_level)** spaces. This is because kramdown (the markdown engine used by Jekyll) indentation for the code block in lists is determined by the column number of the first non-space character after the list item marker. By default, it does not display line numbers. If you want to display line numbers for every code block, you can set `kramdown.syntax_highlighter_opts.block.line_numbers` to true in your `_config.yml` file.

Tensor Tensor::operator+(const Tensor &other) const {
  Tensor out((*this).node->val + other.node->val, {this->node, other.node},
             (*this).node->requires_grad || other.node->requires_grad);
  auto this_node = this->node;
  auto other_node = other.node;
  auto out_node = out.node;
  out_node->backward_fn = [this_node, other_node, out_node]() {
    if (this_node->grad) {
      *(this_node->grad) += *(out_node->grad);
    }
    if (other_node->grad) {
      *(other_node->grad) += *(out_node->grad);
    }
  };
  return out;
}

Tensor Tensor::operator*(const Tensor &other) const {
  Tensor out((*this).node->val * (other.node->val), {this->node, other.node},
             (*this).node->requires_grad || other.node->requires_grad);
  auto this_node = this->node;
  auto other_node = other.node;
  auto out_node = out.node;
  out_node->backward_fn = [this_node, other_node, out_node]() {
    if (this_node->grad) {
      *(this_node->grad) += *(out_node->grad) * other_node->val;
    }
    if (other_node->grad) {
      *(other_node->grad) += *(out_node->grad) * this_node->val;
    }
  };
  return out;
}

Tensor Tensor::matmul(const Tensor &other) const {
  Tensor out((*this).node->val.matmul(other.node->val),
             {this->node, other.node},
             (*this).node->requires_grad || other.node->requires_grad);
  auto this_node = this->node;
  auto other_node = other.node;
  auto out_node = out.node;
  out_node->backward_fn = [this_node, other_node, out_node]() {
    if (this_node->grad) {
      *(this_node->grad) +=
          (*(out_node->grad)).matmul((other_node->val).transpose());
    }
    if (other_node->grad) { // check if exists
      *(other_node->grad) +=
          (this_node->val).transpose().matmul(*(out_node->grad));
    }
  };
  return out;
}

void Tensor::backward() {
  if (!node->requires_grad) {
    throw std::runtime_error("One of the operands doesnt need grad");
  }
  node->grad->fill(1.0); // initial upstream gradient (d(out)/dout = 1)
  std::stack<std::shared_ptr<Node>> st;
  std::vector<std::shared_ptr<Node>> topo;
  st.push(node);
  while (!st.empty()) {
    auto cur = st.top();
    st.pop();
    topo.push_back(cur);
    for (auto &child : cur->child) {
      st.push(child);
    }
  }
  for (auto &node : topo) {
    if (node->backward_fn) {
      node->backward_fn();
    }
  }
}
